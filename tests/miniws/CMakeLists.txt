cmake_minimum_required(VERSION 3.20)
project(miniws C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-Wall -Wextra -MMD -MP -fPIC)
file(GLOB_RECURSE CORE_SRC CONFIGURE_DEPENDS "core/src/*.c")
list(REMOVE_ITEM CORE_SRC "${CMAKE_CURRENT_SOURCE_DIR}/core/src/main.c")
file(GLOB_RECURSE BADE_SRC CONFIGURE_DEPENDS "bade/src/*.c" "bade/manager/*.c")
list(REMOVE_ITEM BADE_SRC "${CMAKE_CURRENT_SOURCE_DIR}/bade/manager/main.c")
set(ARCHIVE "${CMAKE_CURRENT_BINARY_DIR}/modules/core_modules.tar")
file(GLOB_RECURSE MOD_FILES CONFIGURE_DEPENDS "coremod/*")
list(FILTER MOD_FILES EXCLUDE REGEX "/(build|dist|self_archive)/|/(LICENSE|README\\.md)$")
add_custom_command(OUTPUT "${ARCHIVE}"
 COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/modules"
 COMMAND tar -cf "${ARCHIVE}" --exclude=build --exclude=dist --exclude=LICENSE --exclude=README.md --exclude=self_archive -C "${CMAKE_CURRENT_SOURCE_DIR}/coremod" .
 DEPENDS ${MOD_FILES} VERBATIM)
set(EMBED_OBJ "${CMAKE_CURRENT_BINARY_DIR}/modules/core_modules.o")
add_custom_command(OUTPUT "${EMBED_OBJ}"
 COMMAND ld -r -b binary -o "${EMBED_OBJ}" modules/core_modules.tar
 DEPENDS "${ARCHIVE}" WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" VERBATIM)
set_source_files_properties("${EMBED_OBJ}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/build")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build/embedded.h" "#ifndef RUPA_EMBEDDED_H\n#define RUPA_EMBEDDED_H\n#include <stddef.h>\n#include <stdint.h>\n#include <string.h>\n#define EMBED_MODULES_ARCHIVE_NAME \"core_modules.tar\"\n#define EMBED_MODULES_EXTRACT_DIR \"/tmp/core-system\"\n#define EMBED_MODULES_SYMBOL _binary_modules_core_modules_tar_start\n#define EMBED_MODULES_SYMBOL_END _binary_modules_core_modules_tar_end\nextern const unsigned char _binary_modules_core_modules_tar_start[];\nextern const unsigned char _binary_modules_core_modules_tar_end[];\ntypedef struct { const char *key; const char *type; const char *name; const char *extract; size_t size; const uint8_t *data; } Embedded;\nstatic inline const Embedded *getEmbed(const char *key) {\n  static const Embedded e = {\"modules\", \"archive\", EMBED_MODULES_ARCHIVE_NAME, EMBED_MODULES_EXTRACT_DIR, (size_t)(EMBED_MODULES_SYMBOL_END - EMBED_MODULES_SYMBOL), EMBED_MODULES_SYMBOL};\n  return (key && strcmp(key, e.key) == 0) ? &e : NULL;\n}\n#endif\n")
add_library(core_obj OBJECT ${CORE_SRC})
add_library(bade_obj OBJECT ${BADE_SRC})
foreach(t core_obj bade_obj)
 target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" include core/include bade/include core/src bade/src bade/manager)
 set_target_properties(${t} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
add_library(core_lib SHARED $<TARGET_OBJECTS:core_obj>)
add_library(bade_lib SHARED $<TARGET_OBJECTS:bade_obj> "${EMBED_OBJ}")
set_target_properties(core_lib PROPERTIES OUTPUT_NAME core LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
set_target_properties(bade_lib PROPERTIES OUTPUT_NAME bade LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
foreach(t core_lib bade_lib)
 target_link_libraries(${t} PRIVATE ssl crypto m pthread)
endforeach()
add_executable(rupa core/src/main.c)
add_executable(bade bade/manager/main.c)
foreach(t core bade)
 target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" include core/include bade/include core/src bade/src bade/manager)
 target_link_options(${t} PRIVATE -Wl,--no-as-needed)
 target_link_libraries(${t} PRIVATE bade_lib core_lib ssl crypto m pthread)
 set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin" BUILD_RPATH "${CMAKE_CURRENT_BINARY_DIR}/lib")
endforeach()
