cmake_minimum_required(VERSION 3.20)
project(rupaws C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_options(-Wall -Wextra -MMD -MP -fPIC)
file(GLOB_RECURSE RUPA_SRC CONFIGURE_DEPENDS "rupa/src/*.c")
list(REMOVE_ITEM RUPA_SRC "${CMAKE_CURRENT_SOURCE_DIR}/rupa/src/main.c")
file(GLOB_RECURSE RUKA_SRC CONFIGURE_DEPENDS "ruka/src/*.c" "ruka/manager/*.c")
list(REMOVE_ITEM RUKA_SRC "${CMAKE_CURRENT_SOURCE_DIR}/ruka/manager/main.c")
set(ARCHIVE "${CMAKE_CURRENT_BINARY_DIR}/modules/rupa_modules.tar")
file(GLOB_RECURSE MOD_FILES CONFIGURE_DEPENDS "rupamod/*")
list(FILTER MOD_FILES EXCLUDE REGEX "/(build|dist|self_archive)/|/(LICENSE|README\\.md)$")
add_custom_command(OUTPUT "${ARCHIVE}"
 COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/modules"
 COMMAND tar -cf "${ARCHIVE}" --exclude=build --exclude=dist --exclude=LICENSE --exclude=README.md --exclude=self_archive -C "${CMAKE_CURRENT_SOURCE_DIR}/rupamod" .
 DEPENDS ${MOD_FILES} VERBATIM)
set(EMBED_OBJ "${CMAKE_CURRENT_BINARY_DIR}/modules/rupa_modules.o")
add_custom_command(OUTPUT "${EMBED_OBJ}"
 COMMAND ld -r -b binary -o "${EMBED_OBJ}" modules/rupa_modules.tar
 DEPENDS "${ARCHIVE}" WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" VERBATIM)
set_source_files_properties("${EMBED_OBJ}" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/build")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build/embedded.h" "#ifndef RUPA_EMBEDDED_H\n#define RUPA_EMBEDDED_H\n#define EMBED_MODULES_ARCHIVE_NAME \"rupa_modules.tar\"\n#define EMBED_MODULES_EXTRACT_DIR \"/tmp/rupa-system\"\n#define EMBED_MODULES_SYMBOL _binary_modules_rupa_modules_tar_start\n#define EMBED_MODULES_SYMBOL_END _binary_modules_rupa_modules_tar_end\n#endif\n")
add_library(rupa_obj OBJECT ${RUPA_SRC})
add_library(ruka_obj OBJECT ${RUKA_SRC})
foreach(t rupa_obj ruka_obj)
 target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" include rupa/include ruka/include rupa/src ruka/src ruka/manager)
 set_target_properties(${t} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endforeach()
add_library(rupa_lib SHARED $<TARGET_OBJECTS:rupa_obj>)
add_library(ruka_lib SHARED $<TARGET_OBJECTS:ruka_obj> "${EMBED_OBJ}")
set_target_properties(rupa_lib PROPERTIES OUTPUT_NAME rupa LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
set_target_properties(ruka_lib PROPERTIES OUTPUT_NAME ruka LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
foreach(t rupa_lib ruka_lib)
 target_link_libraries(${t} PRIVATE ssl crypto m pthread)
endforeach()
add_executable(rupa rupa/src/main.c)
add_executable(ruka ruka/manager/main.c)
foreach(t rupa ruka)
 target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" include rupa/include ruka/include rupa/src ruka/src ruka/manager)
 target_link_options(${t} PRIVATE -Wl,--no-as-needed)
 target_link_libraries(${t} PRIVATE ruka_lib rupa_lib ssl crypto m pthread)
 set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin" BUILD_RPATH "${CMAKE_CURRENT_BINARY_DIR}/lib")
endforeach()
